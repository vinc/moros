def-fun def

(def (rev ls)
  "Reverses the list"
  (if (nil? ls) ls
    (concat (rev (tail ls)) (cons (head ls) '()))))

def-mac

(def-mac (and @xs)
  (if (nil? xs) true
    (if (nil? (tail xs)) (head xs)
      `(if ,(head xs) (and ,@(tail xs)) false))))

Hypertext Transfer Protocol:

# Connection
(set socket (socket/connect "tcp" (host "moros.cc") 80))

# Request
(set req (str/join (list
  "GET / HTTP/1.1"
  "Host: moros.cc"
  "Connection: close"
  "" "") "\r\n"))
(file/write socket (str->bin req))

# Response
(set mtu (file/size "/dev/net/tcp"))
(set res (bin->str (file/read socket mtu)))
(file/close socket)
(str/trim (first (lines res))) # => "HTTP/1.1 200 OK"
